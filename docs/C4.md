## C4-Container

```mermaid
C4Container
    title MAIA — Containers
    Person(emp, "Employee", "Streamlit chat UI")
    Container_Boundary(m, "MAIA") {
        Container(api, "FastAPI backend", "Python", "/ask, /actions, /actions/confirm")
        Container(rag, "Hybrid retrieval", "Qdrant + BM25 + RRF k=60", "gate >= 0.3, rerank, cite")
        Container(agent, "LangGraph agent", "StateGraph + SQLite ckpt", "slot-fill, interrupt/resume")
        Container(ui, "Streamlit UI", "Python", "chat, approvals")
        ContainerDb(db, "SQLite WAL", "file", "auth, sessions, LTM, checkpoints")
    }
    Rel(emp, ui, "uses", "HTTPS")
    Rel(ui, api, "calls", "HTTP")
    Rel(api, rag, "retrieves", "in-process")
    Rel(api, agent, "executes actions", "in-process")
    Rel(agent, db, "checkpoints", "SQL")
```
